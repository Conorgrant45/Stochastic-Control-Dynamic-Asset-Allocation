\section{Linear Quadratic Regulator (LQR)}
\label{sec:lqr}

We consider the controlled SDE
\[
    dX_s = \bigl[HX_s + M\alpha_s\bigr]\,ds + \sigma\,dW_s,
    \quad s \in [t, T],\quad X_t = x,
\]
and aim to minimise the cost functional
\[
    J^\alpha(t,x) := \E^{t,x}\!\left[
        \int_t^T \!\bigl(X_s^\top C X_s + \alpha_s^\top D\alpha_s\bigr)\,ds
        + X_T^\top R X_T
    \right].
\]
The value function $v(t,x) = \inf_\alpha J^\alpha(t,x)$ satisfies
\[
    v(t,x) = x^\top S(t)x + \int_t^T \tr\!\bigl(\sigma\sigma^\top S(r)\bigr)\,dr,
\]
where $S$ solves the Riccati ODE
\[
    S'(r) = -2H^\top S(r) + S(r)MD^{-1}M^\top S(r) - C,
    \quad S(T) = R,
\]
and the optimal Markov control is $a(t,x) = -D^{-1}M^\top S(t)x$.

\section{Deep Galerkin Method for a Linear PDE}
\label{sec:dgm}

We consider the linear PDE with constant control $\alpha = (1,1)^\top$:
\[
    \partial_t u + \tfrac{1}{2}\tr\!\bigl(\sigma\sigma^\top \partial_{xx}u\bigr)
    + (\partial_x u)^\top Hx + (\partial_x u)^\top M\alpha
    + x^\top Cx + \alpha^\top D\alpha = 0
    \quad \text{on } [0,T)\times\R^2,
\]
\[
    u(T,x) = x^\top Rx \quad \text{on } \R^2.
\]

\subsection{Exercise 3.1 -- Deep Galerkin Approximation}
\label{subsec:ex3.1}

% TODO: describe the DGM residual loss, boundary loss, architecture, training setup.
We approximate $u$ by a neural network $u(\cdot,\cdot;\theta)$ and minimise
\[
    \mathcal{R}(\theta) = \mathcal{R}_{\mathrm{eqn}}(\theta) + \mathcal{R}_{\mathrm{bdry}}(\theta),
\]
where both terms are empirical averages over $N_{\mathrm{batch}}$ randomly sampled
collocation points. Automatic differentiation (via \texttt{torch.autograd}) is used
to compute $\partial_t u$ and $\partial_{xx} u$.

\section{Policy Iteration with DGM}
\label{sec:pia}

\subsection{Algorithm Description}
\label{subsec:pia_algo}

The Policy Improvement Algorithm (PIA) alternates between two steps:

\begin{enumerate}
    \item \textbf{Policy evaluation:} Given the current control network
          $a(\cdot,\cdot;\theta_{\mathrm{act}})$, solve the associated linear PDE
          (as in Section~\ref{sec:dgm}) to update $\theta_{\mathrm{val}}$.
    \item \textbf{Policy improvement:} With $\theta_{\mathrm{val}}$ fixed, update
          $\theta_{\mathrm{act}}$ by minimising the Hamiltonian
          \[
              \mathcal{H}(\theta_{\mathrm{act}}) = \frac{1}{N_{\mathrm{batch}}}
              \sum_{i=1}^{N_{\mathrm{batch}}} \Bigl[
                  (\partial_x v)^\top Hx^{(i)}
                  + (\partial_x v)^\top M\,a(t^{(i)},x^{(i)};\theta_{\mathrm{act}})
                  + (x^{(i)})^\top C x^{(i)}
                  + a^\top D\,a
              \Bigr].
          \]
\end{enumerate}
